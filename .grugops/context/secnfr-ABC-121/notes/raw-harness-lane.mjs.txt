// secnfr-ABC-121 copy of the lane re-hop N8-LANE harness (only S changed) — lane121.mjs (ABC-081's, NFR-008's protocol) with the arm
// list a parameter. The lane is `npx vitest run --dir src` in one fresh OS process per run; the arm
// order rotates every round; ROUNDS rounds with round 0 discarded; counts asserted every run
// (round 0 fixes each arm's expected counts); CPU = user + sys from /usr/bin/time -p.
// Usage: CONDS=A,N,B ROUNDS=9 TAG=x node rehop-lane.mjs     (arms built by rehop-build.mjs)
// Pairs reported: every arm against the FIRST arm, and N against A when both are present.
import { execSync, spawnSync } from "node:child_process";
import { appendFileSync, readFileSync, writeFileSync } from "node:fs";

const S = "/private/tmp/claude-501/-Users-olgeroeselg-Projects-hacks-grugops-examples-cli-chess-example/cc8144c4-cb23-42b5-90b8-44bcbd8a2345/scratchpad/secnfr121";
const ROUNDS = Number(process.env.ROUNDS ?? 9);
const CONDS = (process.env.CONDS ?? "A,N,B").split(",");
const TAG = process.env.TAG ?? "lane";
const OUT = `${S}/${TAG}-raw.tsv`;
const load1 = () => Number(execSync("sysctl -n vm.loadavg").toString().trim().split(/\s+/)[1]);

writeFileSync(OUT, "round\tcond\twall\tuser\tsys\tcpu\tfiles\ttests\tload1_before\tload1_after\tstatus\n");
const expected = {};
for (let round = 0; round < ROUNDS; round++) {
  const k = round % CONDS.length;
  const order = [...CONDS.slice(k), ...CONDS.slice(0, k)];
  for (const c of order) {
    const lb = load1();
    const r = spawnSync("/usr/bin/time", ["-p", "npx", "vitest", "run", "--dir", "src"], { cwd: `${S}/${c}`, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
    const la = load1();
    const out = r.stdout || "";
    const err = r.stderr || "";
    const pick = (key) => { const m = err.match(new RegExp(`^${key}\\s+([0-9.]+)`, "m")); return m ? Number(m[1]) : Number.NaN; };
    const fm = out.match(/Test Files\s+(\d+) passed/);
    const tm = out.match(/^\s*Tests\s+(\d+) passed/m);
    const files = fm ? Number(fm[1]) : Number.NaN;
    const tests = tm ? Number(tm[1]) : Number.NaN;
    if (round === 0) expected[c] = { files, tests };
    const ok = r.status === 0 && files === expected[c].files && tests === expected[c].tests && !/failed/.test(out.split("Test Files")[1] ?? "") ? "ok" : "HARNESS_ERROR";
    const cpu = +(pick("user") + pick("sys")).toFixed(2);
    appendFileSync(OUT, [round, c, pick("real"), pick("user"), pick("sys"), cpu, files, tests, lb, la, ok].join("\t") + "\n");
    console.log(round, c, cpu, pick("real"), files, tests, lb, ok);
  }
}

const rows = readFileSync(OUT, "utf8").trim().split("\n").slice(1).map((l) => {
  const [round, cond, wall, , , cpu, files, tests, lb, , ok] = l.split("\t");
  return { round: +round, cond, wall: +wall, cpu: +cpu, files: +files, tests: +tests, lb: +lb, ok };
});
const med = (xs) => { const s = [...xs].sort((a, b) => a - b); const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };
const use = rows.filter((r) => r.round >= 1);
const lines = [`arms ${CONDS.join(",")} rounds ${ROUNDS} (round 0 discarded)`];
for (const c of CONDS) {
  const rs = use.filter((r) => r.cond === c);
  lines.push(`${c}: n=${rs.length} files/tests=${rs[0].files}/${rs[0].tests} medianCPU=${med(rs.map((r) => r.cpu)).toFixed(3)} rangeCPU=${Math.min(...rs.map((r) => r.cpu))}-${Math.max(...rs.map((r) => r.cpu))} medianWALL=${med(rs.map((r) => r.wall)).toFixed(3)}`);
}
const paired = (x, y) => {
  const d = [];
  for (let round = 1; round < ROUNDS; round++) {
    const a = use.find((r) => r.round === round && r.cond === x);
    const b = use.find((r) => r.round === round && r.cond === y);
    d.push(+(b.cpu - a.cpu).toFixed(2));
  }
  return d;
};
const ref = CONDS[0];
const mRef = med(use.filter((r) => r.cond === ref).map((r) => r.cpu));
const pairs = CONDS.slice(1).map((c) => [ref, c]);
if (CONDS.includes("A") && CONDS.includes("N") && ref !== "A") pairs.push(["A", "N"]);
for (const [x, y] of pairs) {
  const d = paired(x, y);
  const dm = med(d);
  lines.push(`${y} - ${x} CPU per round: ${d.join(" ")} median=${dm.toFixed(3)} (${((100 * dm) / mRef).toFixed(2)}% of ${ref}) sign+ ${d.filter((v) => v > 0).length}/${d.length}`);
}
lines.push(`load1 before each run: min=${Math.min(...rows.map((r) => r.lb))} median=${med(rows.map((r) => r.lb)).toFixed(2)} max=${Math.max(...rows.map((r) => r.lb))}`);
lines.push(`harness errors: ${rows.filter((r) => r.ok !== "ok").length} of ${rows.length}`);
writeFileSync(`${S}/${TAG}-summary.txt`, lines.join("\n") + "\n");
console.log(lines.join("\n"));
