// recut-abc086-lane probe: in-vitest CPU of candidate ABC-086 A1 default-lane slices.
import { it } from "vitest";
import { makeMove, unmakeMove } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/apply.ts";
import { chooseMove } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/engine.ts";
import { parseFen, toFen } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/fen.ts";
import { legalMoves } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/legal.ts";
import { squareName } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/position.ts";
import { virtualClock } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/recut-abc086-lane/src/virtual-clock.ts";
const ROOTS: [string, string][] = [
  ["startpos", "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"],
  ["kiwipete", "r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1"],
  ["position-3", "8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1"],
  ["position-4", "r3k2r/Pppp1ppp/1b3nbN/nP6/BBP1P3/q4N2/Pp1P2PP/R2Q1RK1 w kq - 0 1"],
  ["position-5", "rnbq1k1r/pp1Pbppp/2p5/8/2B5/8/PPP1NnPP/RNBQK2R w KQ - 1 8"],
  ["position-6", "r4rk1/1pp1qppp/p1np1n2/2b1p1B1/2B1P1b1/P1NP1N2/1PP1QPPP/R4RK1 w - - 0 10"],
];
const lbl = (m: any) => squareName(m.from) + squareName(m.to);
const corpus: { name: string; fen: string }[] = [];
for (const [name, fen] of ROOTS) {
  corpus.push({ name, fen });
  const p = parseFen(fen);
  for (const m of legalMoves(p).slice(0, 4)) { const u = makeMove(p, m); corpus.push({ name: name + "/" + lbl(m), fen: toFen(p) }); unmakeMove(p, m, u); }
}
const FAM: Record<string, string[]> = {
  full: ROOTS.map((r) => r[0]),
  sub15: ["startpos", "position-3", "position-5"],
  sub10: ["startpos", "position-3"],
};
it("probe", () => {
  const cond = process.env.COND as string; const budget = Number(process.env.BUDGET ?? 2);
  const fams = FAM[cond]; const slice = corpus.filter((c) => fams.includes(c.name.split("/")[0] as string));
  const rows: string[] = []; let d2 = 0;
  const c0 = process.cpuUsage();
  for (const { name, fen } of slice) {
    const p0 = process.cpuUsage(); const out: string[] = []; let depth = 0;
    for (let r = 0; r < 2; r++) { const c = chooseMove(parseFen(fen), { budgetMs: budget, now: virtualClock(1) }) as any; out.push(lbl(c.move)); depth = c.depth; }
    const pc = process.cpuUsage(p0); if (depth >= 2) d2++;
    rows.push(name + "\td" + depth + "\t" + ((pc.user + pc.system) / 1000).toFixed(1) + "ms\t" + (out[0] === out[1] ? "SAME" : "DIFF"));
  }
  const cu = process.cpuUsage(c0);
  console.log("COND " + cond + " budget " + budget + " n=" + slice.length + "x2 depth>=2: " + d2 + "/" + slice.length + " cpu-ms " + ((cu.user + cu.system) / 1000).toFixed(0) + "\n" + rows.join("\n"));
});
