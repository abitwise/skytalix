// secnfr-ABC-123 probe 1: the N boundary, fixed (HEAD) vs unfixed (e57da89), with a spawn() spy
// preloaded into the WRAPPER process. Target: a node one-liner that prints "ran".
import { spawnSync } from "node:child_process";
const [fixed, unfixed, spy] = process.argv.slice(2);
const TARGET = [process.execPath, "-e", 'process.stdout.write("ran\\n")'];
const USAGE = "usage: node tools/grugops/run-bounded.js <timeoutMs> <command> [args...]\n";
const NS = [
  "2147483646", "2147483647", "2147483647.0", "0x7fffffff", "2.147483647e9",
  "2147483648", "2147483649", "0x80000000", "2.147483648e9", "2147483647.5",
  "4294967296", "4294967297", "99999999999", "9007199254740993", "1e21", "-0", "-2147483648",
];
console.log("N\tunfixed: exit,spawns,target,overflow\tfixed: exit,spawns,target,stderr");
for (const n of NS) {
  const r = (w) => {
    const x = spawnSync(process.execPath, ["--require", spy, w, n, ...TARGET], {
      encoding: "utf8",
      timeout: 10000,
    });
    const spawns = (x.stderr.match(/SPY=(\d+)/) || [])[1];
    const err = x.stderr.replace(/SPY=\d+\n/, "");
    return {
      exit: x.status,
      spawns,
      ran: x.stdout === "ran\n",
      warn: /TimeoutOverflowWarning/.test(err),
      err: err === USAGE ? "usage-only" : err === "" ? "empty" : `OTHER:${JSON.stringify(err.slice(0, 80))}`,
    };
  };
  const u = r(unfixed);
  const f = r(fixed);
  console.log(
    `${JSON.stringify(n)}\t${u.exit},${u.spawns},${u.ran ? "ran" : "-"},${u.warn ? "OVERFLOW" : "-"}\t${f.exit},${f.spawns},${f.ran ? "ran" : "-"},${f.err}`,
  );
}
