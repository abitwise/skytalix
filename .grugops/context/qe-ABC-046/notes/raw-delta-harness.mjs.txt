function once(src, from, to, label) {

  if (n !== 1) throw new Error(`anchor ${label}: found ${n} times`);
  return src.replace(from, to);
}

function vitest(files, env = {}) {
  const r = spawnSync("npx", ["vitest", "run", ...files, "--reporter=json"], { cwd: DIR, encoding: "utf8", env: { ...process.env, ...env }, maxBuffer: 64 * 1024 * 1024, timeout: 600_000 });

  try {
    const j = JSON.parse(o.slice(o.indexOf("{"), o.lastIndexOf("}") + 1));
    const all = j.testResults.flatMap((t) => t.assertionResults.map((a) => ({ file: t.name.replace(`${DIR}/`, ""), name: a.fullName, status: a.status })));
    const perFile = {};
    for (const a of all) { perFile[a.file] ??= { total: 0, failed: 0 }; perFile[a.file].total++; if (a.status === "failed") perFile[a.file].failed++; }
    return { status: r.status, total: j.numTotalTests, passed: j.numPassedTests, failed: j.numFailedTests, skipped: j.numPendingTests + j.numTodoTests, perFile, failedNames: all.filter((a) => a.status === "failed").map((a) => `${a.file} :: ${a.name}`), all };
  } catch (e) { return { status: r.status, parseError: String(e), tail: (r.stdout + r.stderr).slice(-1500) }; }
}
