// Preload: counts child_process.spawn calls made by the wrapper, reports the count at exit on fd 3-free stderr tag.
const cp = require("node:child_process");
const { syncBuiltinESMExports } = require("node:module");
const real = cp.spawn;
let calls = 0;
cp.spawn = (...a) => { calls++; return real(...a); };
syncBuiltinESMExports();
process.on("exit", () => { require("node:fs").writeSync(2, `SPY spawn() calls: ${calls}\n`); });
