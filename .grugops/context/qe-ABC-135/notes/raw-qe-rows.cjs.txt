// QE ABC-135: adversarial comment spellings against the base and tip instruments and a TS oracle.
// usage: node qe-probe.cjs <base render-seam.test.ts> <tip render-seam.test.ts>
const ts = require("typescript");
const fs = require("node:fs");
const lift = (file) => {
  const text = fs.readFileSync(file, "utf8");
  const sf = ts.createSourceFile(file, text, ts.ScriptTarget.Latest, true);
  const decls = [];
  for (const st of sf.statements)
    if (ts.isVariableStatement(st))
      for (const d of st.declarationList.declarations)
        if (["withoutComments", "bindsAValueFrom"].includes(d.name.getText(sf))) decls.push(st.getText(sf));
  const js = ts.transpileModule(`${decls.join("\n")}\nmodule.exports = bindsAValueFrom;`, { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022 } }).outputText;
  const m = { exports: {} };
  new Function("module", "exports", js)(m, m.exports);
  return m.exports;
};
const oracle = (source, target) => ts.createSourceFile("x.ts", source, ts.ScriptTarget.Latest, true).statements.some((st) => {
  if (!ts.isImportDeclaration(st) || st.moduleSpecifier.text !== `./${target}`) return false;
  const c = st.importClause;
  if (!c || c.isTypeOnly) return false;
  if (c.name) return true;
  const nb = c.namedBindings;
  if (!nb) return false;
  return ts.isNamespaceImport(nb) || nb.elements.some((e) => !e.isTypeOnly);
});
const [base, tip] = process.argv.slice(2).map(lift);
const probes = [
["R1 { /* c */ type X }", 'import { /* c */ type X } from "./m.ts";'],
["R2 all-type + trailing // c", 'import {\n  type X, // c\n} from "./m.ts";'],
["R3 all-type + // f, line", 'import {\n  type X,\n  // f,\n} from "./m.ts";'],
["R4 import in /* */ block", '/*\nimport { f } from "./m.ts";\n*/'],
["R5 multi-line mixed brace", 'import {\n  type X,\n  f,\n} from "./m.ts";'],
["R6 type then value statement", 'import type { X } from "./m.ts";\nimport { f } from "./m.ts";'],
];
for (const [name, src] of probes) {
  const b = base(src, "m.ts"), t = tip(src, "m.ts"), o = oracle(src, "m.ts");
  const tag = t === o ? (b === o ? "ok" : "FIXED") : b === t ? "WRONG-unchanged" : t ? "WRONG-overcount-NEW" : "WRONG-undercount-NEW";
  console.log(`${name.padEnd(46)} base=${String(b).padEnd(5)} tip=${String(t).padEnd(5)} oracle=${String(o).padEnd(5)} ${tag}`);
}
