// are written back and the sha256 asserted equal. Usage: node mutants.mjs <worktree>

import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";

const CI = `${DIR}/src/command-input.ts`;

const sha = (s) => createHash("sha256").update(s).digest("hex");

const pristineSha = Object.fromEntries(Object.entries(pristine).map(([k, v]) => [k, sha(v)]));
const loadavg = () => execSync("sysctl -n vm.loadavg").toString().trim();

function once(src, from, to, label) {
  const n = src.split(from).length - 1;
  if (n !== 1) throw new Error(`anchor ${label}: found ${n} times`);
  return src.replace(from, to);
}

const ARM = `quitArmed = classifyCommand(line)?.token === "quit";`;
const DISARM = `    disarm: (): void => {
      quitArmed = false;
    },`;
const READER_CALL = `      resolve.disarm();\n`;
const M9_RET = `    return resolve(history, line.text);`;
const M9_IMPORT = `import { createPromptLineResolver } from "./command-input.ts";`;

const MUTANTS = {

  Q3: { ci: (s) => once(s, ARM, `quitArmed = line.trim().toLowerCase().startsWith("quit");`, "arm") },
  X1: { ci: (s) => once(s, DISARM, `    disarm: (): void => {
      /* no-op */

  X2: { li: (s) => once(s, READER_CALL, "", "reader-call") },
  X3: { ci: (s) => once(s, DISARM, `    disarm: (): void => {
      quitArmed = true;
    },`, "disarm") },
  M9: {

function vitest(files, env = {}) {
  const r = spawnSync("npx", ["vitest", "run", ...files, "--reporter=json"], {
    cwd: DIR, encoding: "utf8", env: { ...process.env, ...env }, maxBuffer: 64 * 1024 * 1024, timeout: 600_000,
  });
  const out = r.stdout;
  try {
    const j = JSON.parse(out.slice(out.indexOf("{"), out.lastIndexOf("}") + 1));
    const all = j.testResults.flatMap((t) => t.assertionResults.map((a) => ({ file: t.name.replace(`${DIR}/`, ""), name: a.fullName, status: a.status })));
    return {
      status: r.status, total: j.numTotalTests, passed: j.numPassedTests, failed: j.numFailedTests,
      skipped: j.numPendingTests + j.numTodoTests,
      failedNames: all.filter((a) => a.status === "failed").map((a) => `${a.file} :: ${a.name}`),
      all,
    };
  } catch (e) {
    return { status: r.status, parseError: String(e), tail: (r.stdout + r.stderr).slice(-1500) };
  }
}
