/**
 * secnfr-ABC-121 probe — NFR-004 (3 s / 5 s ceiling), NFR-005 (depth floor 6 on plain node),
 * NFR-009 (16 MiB, growth <= 1 MiB). Plain `node`, importing the reviewed tip's src/ read-only.
 *   node nfr-probe.ts time [rounds]     — six benchmark roots, chooseMove at DEFAULT_BUDGET_MS,
 *                                          outer wall clock + elapsedMs + depth; kiwipete/position-6
 *                                          also on the pre-ABC-121 door; one 10 s-budget ceiling call
 *   node --expose-gc nfr-probe.ts mem   — per position: gc'd reads after a 1 s, a 3 s and a
 *                                          second 3 s chooseMove
 */
import { loadavg } from "node:os";
import { performance } from "node:perf_hooks";
import {
  chooseMove,
  chooseMoveWithoutPvsOrReductions,
  DEFAULT_BUDGET_MS,
} from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/ABC-121-pvs-lmr/src/engine.ts";
import { parseFen } from "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/ABC-121-pvs-lmr/src/fen.ts";

const POS: Array<[string, string]> = [
  ["startpos", "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"],
  ["kiwipete", "r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1"],
  ["position-3", "8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1"],
  ["position-4", "r3k2r/Pppp1ppp/1b3nbN/nP6/BBP1P3/q4N2/Pp1P2PP/R2Q1RK1 w kq - 0 1"],
  ["position-5", "rnbq1k1r/pp1Pbppp/2p5/8/2B5/8/PPP1NnPP/RNBQK2R w KQ - 1 8"],
  ["position-6", "r4rk1/1pp1qppp/p1np1n2/2b1p1B1/2B1P1b1/P1NP1N2/1PP1QPPP/R4RK1 w - - 0 10"],
];
const mode = process.argv[2];

if (mode === "time") {
  const rounds = Number(process.argv[3] ?? 3);
  for (let round = 1; round <= rounds; round++) {
    for (const [name, fen] of POS) {
      const arms: Array<[string, typeof chooseMove]> = [["chooseMove", chooseMove]];
      if (name === "kiwipete" || name === "position-6") arms.push(["door-pre121", chooseMoveWithoutPvsOrReductions]);
      if (round % 2 === 0) arms.reverse();
      for (const [arm, choose] of arms) {
        const load1 = loadavg()[0]?.toFixed(2);
        const t0 = performance.now();
        const c = choose(parseFen(fen), { budgetMs: DEFAULT_BUDGET_MS });
        const wall = performance.now() - t0;
        if (c === null) throw new Error("no move");
        console.log(JSON.stringify({ round, name, arm, depth: c.depth, nodes: c.nodes, elapsedMs: c.elapsedMs, wallMs: +wall.toFixed(1), load1 }));
      }
    }
  }
  const t0 = performance.now();
  const c = chooseMove(parseFen(POS[1]![1]), { budgetMs: 10_000 });
  console.log(JSON.stringify({ ceiling: true, name: "kiwipete", budgetMs: 10_000, depth: c?.depth, elapsedMs: c?.elapsedMs, wallMs: +(performance.now() - t0).toFixed(1) }));
} else if (mode === "mem") {
  const g = (globalThis as { gc?: () => void }).gc;
  if (!g) throw new Error("run with --expose-gc");
  const read = () => { g(); g(); const m = process.memoryUsage(); return { heapUsed: m.heapUsed, arrayBuffers: m.arrayBuffers }; };
  const mib = (n: number) => +(n / 1048576).toFixed(3);
  for (const [name, fen] of POS.filter(([n]) => n === "kiwipete" || n === "position-6" || n === "position-3")) {
    const out: Record<string, unknown> = { name };
    for (const [label, ms] of [["1s", 1000], ["3s", 3000], ["3s-again", 3000]] as const) {
      const c = chooseMove(parseFen(fen), { budgetMs: ms });
      const r = read();
      out[label] = { depth: c?.depth, heapUsedMiB: mib(r.heapUsed), arrayBuffersMiB: mib(r.arrayBuffers) };
    }
    const a = out["1s"] as { heapUsedMiB: number; arrayBuffersMiB: number };
    const b = out["3s"] as { heapUsedMiB: number; arrayBuffersMiB: number };
    const b2 = out["3s-again"] as { heapUsedMiB: number; arrayBuffersMiB: number };
    out.growth1to3 = { heapUsedMiB: +(b.heapUsedMiB - a.heapUsedMiB).toFixed(3), arrayBuffersMiB: +(b.arrayBuffersMiB - a.arrayBuffersMiB).toFixed(3) };
    out.growth3to3 = { heapUsedMiB: +(b2.heapUsedMiB - b.heapUsedMiB).toFixed(3), arrayBuffersMiB: +(b2.arrayBuffersMiB - b.arrayBuffersMiB).toFixed(3) };
    console.log(JSON.stringify(out));
  }
} else {
  throw new Error("mode: time | mem");
}
