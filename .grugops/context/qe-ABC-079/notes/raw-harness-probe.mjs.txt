// QE ABC-079 probe: the SHIPPED search() on the six perft roots at depths 1-3, in plain node,
// against whatever src/ currently holds. Used clean (the reference) and under each mutant, to
// show whether the mutant is a REAL defect (score/move changed) or cost-only (nodes changed).
// Usage: node probe.mjs <out.json> [maxDepth]
import { writeFileSync } from "node:fs";

const WT = "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/agent-ae4e87bedf6be5ad4";
const { parseFen } = await import(`${WT}/src/fen.ts`);
const { search } = await import(`${WT}/src/search.ts`);
const { squareName } = await import(`${WT}/src/position.ts`);
const PERFT = [
  ["startpos", "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"],
  ["kiwipete", "r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1"],
  ["position-3", "8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1"],
  ["position-4", "r3k2r/Pppp1ppp/1b3nbN/nP6/BBP1P3/q4N2/Pp1P2PP/R2Q1RK1 w kq - 0 1"],
  ["position-5", "rnbq1k1r/pp1Pbppp/2p5/8/2B5/8/PPP1NnPP/RNBQK2R w KQ - 1 8"],
  ["position-6", "r4rk1/1pp1qppp/p1np1n2/2b1p1B1/2B1P1b1/P1NP1N2/1PP1QPPP/R4RK1 w - - 0 10"],
];
const maxDepth = Number(process.argv[3] ?? 3);
const out = {};
for (const [name, fen] of PERFT) {
  out[name] = {};
  for (let d = 1; d <= maxDepth; d++) {
    const t0 = Date.now();
    const r = search(parseFen(fen), d);
    out[name][`d${d}`] = { score: r.score, move: r.move ? `${squareName(r.move.from)}${squareName(r.move.to)}` : null, nodes: r.nodes, ms: Date.now() - t0 };
  }
}
writeFileSync(process.argv[2], JSON.stringify(out, null, 1));
console.log(JSON.stringify(out));
