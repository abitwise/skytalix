function restore() {
  spawnSync("git", ["checkout", "--", "src/command-input.ts", "src/turn-loop.ts"], { cwd: CWD });
  const ok = sha(readFileSync(CI, "utf8")) === sha(ORIGINAL[CI]) && sha(readFileSync(TL, "utf8")) === sha(ORIGINAL[TL]);
  const status = spawnSync("git", ["status", "--porcelain", "--", "src"], { cwd: CWD, encoding: "utf8" }).stdout.trim();
  return { restoredByteIdentical: ok, srcStatusEmpty: status === "" , status };
}
