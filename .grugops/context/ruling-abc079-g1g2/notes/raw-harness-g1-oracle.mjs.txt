// G1 probe: is the shipped depth-3 {score, move} equal to the retired unpruned QUIESCENCE oracle?
// Plain node, clean worktree src/. Usage: node g1-oracle.mjs <srcDir> <names...>
import { loadavg } from "node:os";
const SRC = process.argv[2];
const { parseFen } = await import(`${SRC}/fen.ts`);
const { squareName } = await import(`${SRC}/position.ts`);
const { search, searchWithoutPruning } = await import(`${SRC}/search.ts`);
const CORPUS = {
  startpos: "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1",
  kiwipete: "r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1",
  "position-3": "8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1",
  "position-4": "r3k2r/Pppp1ppp/1b3nbN/nP6/BBP1P3/q4N2/Pp1P2PP/R2Q1RK1 w kq - 0 1",
  "position-5": "rnbq1k1r/pp1Pbppp/2p5/8/2B5/8/PPP1NnPP/RNBQK2R w KQ - 1 8",
  "position-6": "r4rk1/1pp1qppp/p1np1n2/2b1p1B1/2B1P1b1/P1NP1N2/1PP1QPPP/R4RK1 w - - 0 10",
};
const lab = (m) => (m ? `${squareName(m.from)}${squareName(m.to)}${m.promotion ? `=${m.promotion}` : ""}` : null);
for (const name of process.argv.slice(3)) {
  const l0 = loadavg()[0].toFixed(2);
  let t = performance.now();
  const p = search(parseFen(CORPUS[name]), 3);
  const pm = Math.round(performance.now() - t);
  t = performance.now();
  const u = searchWithoutPruning(parseFen(CORPUS[name]), 3);
  const um = Math.round(performance.now() - t);
  console.log(JSON.stringify({ name, depth: 3, shipped: { score: p.score, move: lab(p.move), nodes: p.nodes, ms: pm },
    oracle: { score: u.score, move: lab(u.move), nodes: u.nodes, ms: um },
    scoreEqual: Object.is(p.score, u.score), moveEqual: lab(p.move) === lab(u.move), load: [l0, loadavg()[0].toFixed(2)] }));
}
