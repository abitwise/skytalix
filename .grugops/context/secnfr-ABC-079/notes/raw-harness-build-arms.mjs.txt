// secnfr-ABC-079: build the three NFR-008 lane arms outside the repository.
// A = cb5540d as delivered; N = byte-identical copy (null control); B = cb5540d with ABC-079's
// seven src files restored to 4f0ef01 (quiescence.ts / quiescence.test.ts absent there).
import { execFileSync } from "node:child_process";
import { mkdirSync, rmSync, symlinkSync, writeFileSync } from "node:fs";

const S = "/private/tmp/claude-501/-Users-olgeroeselg-Projects-hacks-grugops-examples-cli-chess-example/cc8144c4-cb23-42b5-90b8-44bcbd8a2345/scratchpad/lane";
const W = "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/secnfr-ABC-079";
const NM = "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/node_modules";
const git = (...a) => execFileSync("git", ["-C", W, ...a], { maxBuffer: 256 * 1024 * 1024 });

rmSync(S, { recursive: true, force: true });
for (const arm of ["A", "N", "B"]) {
  mkdirSync(`${S}/${arm}`, { recursive: true });
  const tar = git("archive", "cb5540d");
  execFileSync("tar", ["-x", "-C", `${S}/${arm}`], { input: tar });
  symlinkSync(NM, `${S}/${arm}/node_modules`);
}
rmSync(`${S}/B/src/quiescence.ts`);
rmSync(`${S}/B/src/quiescence.test.ts`);
for (const f of ["search.ts", "search.test.ts", "search.qe.test.ts", "engine.ts", "engine.test.ts"]) {
  writeFileSync(`${S}/B/src/${f}`, git("show", `4f0ef01:src/${f}`));
}
