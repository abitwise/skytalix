const count = (hay, needle) => hay.split(needle).length - 1;
const vitest = (name, files) => {
  const load = os.loadavg().map((x) => x.toFixed(2)).join(" ");
  const t0 = process.hrtime.bigint();
  const r = spawnSync("npx", ["vitest", "run", ...files, "--reporter=json"], { cwd: WT, encoding: "utf8", maxBuffer: 64 << 20 });
  const secs = (Number(process.hrtime.bigint() - t0) / 1e9).toFixed(2);
  writeFileSync(`${LOGS}/mutant-${name}.json`, r.stdout);
  writeFileSync(`${LOGS}/mutant-${name}.stderr.log`, r.stderr ?? "");
  let j;
  try { j = JSON.parse(r.stdout.slice(r.stdout.indexOf("{"))); } catch { return { exit: r.status, parse: "FAILED", secs, load, stderrTail: (r.stderr ?? "").slice(-600) }; }
  const failedRows = [];
  for (const f of j.testResults ?? []) for (const a of f.assertionResults ?? []) if (a.status === "failed") failedRows.push({ row: a.fullName, msg: (a.failureMessages?.[0] ?? "").split("\n")[0].slice(0, 160) });
  return { exit: r.status, secs, load, passed: j.numPassedTests, failed: j.numFailedTests, total: j.numTotalTests, failedRows };
};

const original = readFileSync(FILE, "utf8");
const results = {};
const clean = () => git("status", "--porcelain", "--", "src").trim();
if (clean() !== "") throw new Error(`src not clean before the round:\n${clean()}`);

results.M0_control = vitest("M0_control", ["src/turn-loop.test.ts", "src/turn-loop.nfr.test.ts"]);
console.log("M0_control", JSON.stringify(results.M0_control));

for (const [name, edits] of Object.entries(MUTANTS)) {
  let text = original;
  for (const { find, replace } of edits) {
    const n = count(text, find);
    if (n !== 1) throw new Error(`${name}: needle matched ${n} times:\n${find}`);
    text = text.replace(find, replace);
  }
  writeFileSync(FILE, text);
  const inFile = vitest(name, ["src/turn-loop.test.ts", "src/turn-loop.nfr.test.ts"]);
  let lane;
  if (inFile.failed === 0) lane = vitest(`${name}-lane`, ["--dir", "src"]);
  git("checkout", "--", "src/turn-loop.ts");
  const after = clean();
  const restored = readFileSync(FILE, "utf8") === original;
  results[name] = { inFile, lane, revertedClean: after === "" && restored };
  console.log(name, JSON.stringify({ failed: inFile.failed, passed: inFile.passed, rows: inFile.failedRows?.map((r) => r.row.replace(/^.*> /, "").slice(0, 60)), lane: lane ? { failed: lane.failed, passed: lane.passed } : undefined, revertedClean: results[name].revertedClean, secs: inFile.secs, load: inFile.load }));
  if (!results[name].revertedClean) throw new Error(`${name}: revert left src dirty`);
}
