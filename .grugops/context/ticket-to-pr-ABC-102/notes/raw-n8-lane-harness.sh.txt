#!/bin/bash
# usage: n8.sh <after-dir> <base-dir> <infile-n> <lane-n>
A=$1; B=$2
run() { # dir label args...
  local d=$1 l=$2; shift 2
  local load=$(sysctl -n vm.loadavg | awk '{print $2}')
  local out=$( cd $d && { /usr/bin/time -p npx vitest run "$@" ; } 2>&1 )
  local u=$(echo "$out" | awk '/^user/{print $2}') s=$(echo "$out" | awk '/^sys/{print $2}')
  local t=$(echo "$out" | grep -oE 'tests [0-9]+ms' | head -1)
  local c=$(echo "$out" | grep -E '^ +Tests ' | head -1 | tr -s ' ')
  echo "$l load=$load user=$u sys=$s cpu=$(echo "$u+$s"|bc) $t |$c"
}
for i in $(seq 1 $3); do
  if (( i % 2 )); then run $B before src/command-input.test.ts; run $A after src/command-input.test.ts
  else run $A after src/command-input.test.ts; run $B before src/command-input.test.ts; fi
done
for i in $(seq 1 $4); do
  if (( i % 2 )); then run $B lane-before --dir src; run $A lane-after --dir src
  else run $A lane-after --dir src; run $B lane-before --dir src; fi
done
