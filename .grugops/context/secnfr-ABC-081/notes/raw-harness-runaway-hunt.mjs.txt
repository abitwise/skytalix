// secnfr-ABC-081 security probe, part 2: hunt for a board whose WHOLE tree dies out (every line
// ends in an in-tree repetition or a terminal node), so each deeper iteration is ~free and
// chooseMove's uncapped deepening loop (`for (iteration = 2; now() < deadline; iteration++)`)
// runs to a huge depth. Stakes: KEY_ARRAY_CAPACITY (128) and the packed word's 13-bit depth
// field (`depth << 19` in a Uint32 — depth >= 8192 does not fit).
// Random kings + a locked pawn structure; keep boards where BOTH sides have <= 2 legal moves,
// not in check, not game-over; run chooseMove at a short budget and report the depth reached.
// argv: armsDir arm budgetMs samples
const [ARMS, ARM, BUDGET, SAMPLES] = process.argv.slice(2);
const R = `${ARMS}/${ARM}`;
const { parseFen } = await import(`${R}/src/fen.ts`);
const { chooseMove } = await import(`${R}/src/engine.ts`);
const { legalMoves, isInCheck } = await import(`${R}/src/legal.ts`);

// Pawn skeletons (rank 8 first), locked files, kings to be dropped on empty squares.
const SKELETONS = [
  "8/8/8/p1p1p1p1/P1P1P1P1/8/8/8",
  "8/8/p7/P7/8/8/8/8",
  "8/8/8/1p6/1P6/8/8/8",
  "8/p7/P7/8/8/8/8/8",
  "8/8/8/8/8/p7/P7/8",
  "8/1p6/1P6/8/8/8/8/8",
  "8/pp6/PP6/8/8/8/8/8",
  "8/8/8/8/8/pp6/PP6/8",
];
function expand(sk) {
  const board = [];
  for (const row of sk.split("/")) {
    const r = [];
    for (const ch of row) {
      if (/\d/.test(ch)) for (let i = 0; i < Number(ch); i++) r.push(null);
      else r.push(ch);
    }
    board.push(r);
  }
  return board;
}
function fenOf(board, stm) {
  return `${board
    .map((r) => {
      let s = "";
      let e = 0;
      for (const c of r) {
        if (c === null) e++;
        else {
          if (e) s += e;
          e = 0;
          s += c;
        }
      }
      return s + (e ? e : "");
    })
    .join("/")} ${stm} - - 0 1`;
}
const opp = (fen) => fen.replace(/ ([wb]) /, (_, s) => ` ${s === "w" ? "b" : "w"} `);
let seed = 12345;
const rnd = (n) => {
  seed = (seed * 1103515245 + 12345) & 0x7fffffff;
  return seed % n;
};
const hits = [];
const seen = new Set();
const cands = [];
let tried = 0;
for (let i = 0; i < Number(SAMPLES); i++) {
  const sk = SKELETONS[rnd(SKELETONS.length)];
  const b = expand(sk);
  const free = [];
  for (let r = 0; r < 8; r++) for (let f = 0; f < 8; f++) if (b[r][f] === null) free.push([r, f]);
  const a = free[rnd(free.length)];
  let c = free[rnd(free.length)];
  if (a === c || (Math.abs(a[0] - c[0]) <= 1 && Math.abs(a[1] - c[1]) <= 1)) continue;
  b[a[0]][a[1]] = "K";
  b[c[0]][c[1]] = "k";
  const fen = fenOf(b, "w");
  let p;
  let q;
  try {
    p = parseFen(fen);
    q = parseFen(opp(fen));
  } catch {
    continue;
  }
  if (isInCheck(p) || isInCheck(q)) continue;
  const nw = legalMoves(p).length;
  const nb = legalMoves(q).length;
  if (nw === 0 || nb === 0 || nw > 2 || nb > 2) continue;
  if (seen.has(fen)) continue;
  seen.add(fen);
  cands.push({ fen, nw, nb });
}
// Fewest moves first; evaluate at most 60 so the probe is bounded.
cands.sort((x, y) => x.nw * x.nb - y.nw * y.nb);
for (const c of cands.slice(0, 60)) {
  tried++;
  const ch = chooseMove(parseFen(c.fen), { budgetMs: Number(BUDGET) });
  hits.push({ ...c, depth: ch.depth, nodes: ch.nodes, score: ch.score });
}
hits.sort((x, y) => y.depth - x.depth);
console.log(JSON.stringify({ arm: ARM, samples: Number(SAMPLES), distinctCandidates: cands.length, evaluated: tried, top: hits.slice(0, 8) }, null, 1));
