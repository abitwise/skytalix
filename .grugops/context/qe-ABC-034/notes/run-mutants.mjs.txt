const git = (args) => spawnSync("git", args, { cwd: W, encoding: "utf8" });
const results = [];
for (const m of MUTANTS) {
  if (only.length && !only.includes(m.id)) continue;
  const original = readFileSync(m.file, "utf8");
  const count = original.split(m.from).length - 1;
  if (count !== 1) {
    console.log(`[${m.id}] ANCHOR MATCHED ${count} TIMES — skipped`);
    results.push({ id: m.id, what: m.what, red: "ANCHOR-NOT-FOUND", failed: [] });
    continue;
  }
  writeFileSync(m.file, original.replace(m.from, m.to));
  const t0 = Date.now();
  const r = spawnSync("npx", ["vitest", "run", "--dir", "src", "--reporter=json"], { cwd: W, encoding: "utf8", maxBuffer: 1 << 28 });
  const ms = Date.now() - t0;
  writeFileSync(`${S}/mutant-${m.id}.json.txt`, r.stdout + "\n--- STDERR ---\n" + r.stderr);
  let failed = [];
  let summary = "";
  try {
    const j = JSON.parse(r.stdout.slice(r.stdout.indexOf("{")));
    summary = `files=${j.numTotalTestSuites} total=${j.numTotalTests} passed=${j.numPassedTests} failed=${j.numFailedTests} skip=${j.numPendingTests + j.numTodoTests}`;
    for (const f of j.testResults) {
      for (const a of f.assertionResults) {
        if (a.status === "failed") failed.push(`${f.name.replace(W + "/", "")} :: ${a.fullName}`);
      }
    }
  } catch (e) {
    summary = `JSON PARSE FAILED exit=${r.status} (tsc/syntax?) stderr head: ${r.stderr.slice(0, 300)}`;
  }
  // revert and verify clean
  git(["checkout", "--", m.file]);
  const after = readFileSync(m.file, "utf8");
  const clean = after === original ? "REVERTED-CLEAN" : "REVERT-FAILED";
  const st = git(["status", "--short", "--", "src/"]).stdout.trim();
  console.log(`[${m.id}] ${m.what}\n   exit=${r.status} ${ms}ms ${summary}\n   RED rows: ${failed.length === 0 ? "NONE" : failed.length}\n   ${clean}; git status src/: ${st === "" ? "(clean)" : st}`);
  for (const f of failed) console.log(`     - ${f}`);
  results.push({ id: m.id, what: m.what, summary, exit: r.status, red: failed.length, failed, clean, status: st });
}
writeFileSync(`${S}/mutant-results.json`, JSON.stringify(results, null, 2));
