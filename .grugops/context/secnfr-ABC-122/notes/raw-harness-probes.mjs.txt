function alive(pid) {
  try { process.kill(pid, 0); return true; } catch (e) { if (e.code === "ESRCH") return false; if (e.code === "EPERM") return true; throw e; }
}
import { execFileSync } from "node:child_process";
// Kills ONLY processes whose command line is one of this probe's own (wrapper, fixture, or the
// probe's inline -e children), so a reused pid can never be hit. Groups are expanded to members.
const MINE = [WRAPPER_MARK(), "run-bounded.hang-fixture.js", "setInterval(()=>{},1000)", "setInterval(() => {}, 1000)", "setTimeout(()=>{},300)"];
function WRAPPER_MARK() { return "tools/grugops/run-bounded.js"; }
function members(pid) {
  const rows = execFileSync("ps", ["-axo", "pid=,pgid=,command="]).toString().trim().split("\n").map((l) => { const m = l.trim().match(/^(\d+)\s+(\d+)\s+(.*)$/); return m ? { pid: +m[1], pgid: +m[2], cmd: m[3] } : null; }).filter(Boolean);
  const pick = pid < 0 ? rows.filter((r) => r.pgid === -pid) : rows.filter((r) => r.pid === pid);
  return pick.filter((r) => MINE.some((k) => r.cmd.includes(k)));
}
function kill(pid, sig = "SIGKILL") { for (const r of members(pid)) { try { process.kill(r.pid, sig); } catch {} } }
// signal() is the PROBE STIMULUS (sent to a pid this probe just spawned and is watching live).
function signal(pid, sig) { try { process.kill(pid, sig); } catch {} }

// Runs the wrapper as the leader of ITS OWN group (detached), so the probe can model an outer
// supervisor that kills by group, and can always sweep it.
function runWrapper(args, { onLine, env } = {}) {
  const started = performance.now();
  const w = spawn(NODE, [WRAPPER, ...args], { detached: true, stdio: ["ignore", "pipe", "pipe"], env: env ?? process.env });
  seen.add(w.pid); seen.add(-w.pid);
  let out = ""; let err = "";
  w.stdout.on("data", (c) => { out += c; if (onLine && out.includes("\n")) { const l = out.split("\n")[0]; if (!w._fired) { w._fired = true; onLine(JSON.parse(l), w); } } });
  w.stderr.on("data", (c) => { err += c; });
  const done = new Promise((resolve) => {
    const outer = setTimeout(() => { kill(-w.pid); kill(w.pid); resolve({ code: "OUTER_BOUND", signal: null }); }, PROBE_BOUND_MS);
    w.on("exit", (code, signal) => { clearTimeout(outer); resolve({ code, signal }); });
  });
  return { w, started, done: done.then((r) => ({ ...r, ms: +(performance.now() - started).toFixed(1), out, err: err.trim() })) };
}

// ---------- 4. signals delivered to the WRAPPER while the fixture tree hangs ----------
// Each case: wrapper at a 4 s bound around the hang fixture (fixture + grandchild in the child's
// new session/group). Signal the wrapper 300 ms after the fixture reports its pids, then look at
// what is alive 500 ms after the wrapper is gone.
for (const [label, how] of [
  ["SIGINT to wrapper pid", (w) => signal(w.pid, "SIGINT")],
  ["SIGTERM to wrapper pid", (w) => signal(w.pid, "SIGTERM")],
  ["SIGHUP to wrapper pid (terminal closed)", (w) => signal(w.pid, "SIGHUP")],
  ["SIGQUIT to wrapper pid (Ctrl-\\)", (w) => signal(w.pid, "SIGQUIT")],
  ["SIGKILL to wrapper pid", (w) => signal(w.pid, "SIGKILL")],
  ["outer supervisor SIGTERM to the wrapper's GROUP", (w) => signal(-w.pid, "SIGTERM")],
  ["outer supervisor SIGKILL to the wrapper's GROUP", (w) => signal(-w.pid, "SIGKILL")],
]) {
  let pids = null;
  const { w, done } = runWrapper(["4000", NODE, FIXTURE], {
    onLine: (p) => { pids = p; seen.add(p.fixture); seen.add(-p.fixture); seen.add(p.grandchild); setTimeout(() => how(w), 300); },
  });
  const r = await done;
  await sleep(500);
  const armed = pids !== null;
  const survivors = armed ? [["fixture", pids.fixture], ["grandchild", pids.grandchild], ["group", -pids.fixture]].filter(([, p]) => alive(p) && members(p).length > 0).map(([n]) => n) : ["unarmed"];
  rec(`signal: ${label}`, survivors.length === 0, { wrapperExit: r.code, wrapperSignal: r.signal, ms: r.ms, survivors, stderr: r.err.slice(0, 140) });
  if (armed) {

  }
}
