const BASE = "20ad7cb";
const git = (...a) => execFileSync("git", a, { cwd: WT, encoding: "utf8", maxBuffer: 64 << 20 });
const sha = (s) => createHash("sha256").update(s).digest("hex").slice(0, 16);
const show = (rev, path) => git("show", `${rev}:${path}`);
const head = git("rev-parse", "--short", "HEAD").trim();

const baseTs = show(BASE, "src/turn-loop.ts").split("\n");
const headTs = show("HEAD", "src/turn-loop.ts").split("\n");
const baseTest = show(BASE, "src/turn-loop.test.ts").split("\n");

// Extract a block by a start regex and an end regex (inclusive), from a line array.
const block = (lines, startRe, endRe, from = 0) => {
  const s = lines.findIndex((l, i) => i >= from && startRe.test(l));
  if (s < 0) throw new Error(`start not found: ${startRe}`);
  const e = lines.findIndex((l, i) => i >= s && endRe.test(l));
  if (e < 0) throw new Error(`end not found: ${endRe}`);
  return { text: lines.slice(s, e + 1).join("\n"), range: `${s + 1}-${e + 1}` };
};
const stripComments = (text) =>
  text
    .split("\n")
    .filter((l) => {
      const t = l.trim();
      return t !== "" && !t.startsWith("*") && !t.startsWith("/**") && !t.startsWith("//") && !t.startsWith("*/");
    })
    .map((l) => l.replace(/\s+\/\/.*$/, ""))
    .join("\n");

const out = { head, base: BASE };
// A7 — TerminalReason type, LoopOutcome body (comments stripped), INPUT_ENDED_NOTICE literal.
const cmp = (name, b, h, strip = false) => {
  const bt = strip ? stripComments(b.text) : b.text;
  const ht = strip ? stripComments(h.text) : h.text;
  out[name] = { base: `${b.range} ${sha(bt)}`, head: `${h.range} ${sha(ht)}`, identical: bt === ht };
};

out.A7_INPUT_ENDED_NOTICE = { base: baseTs.find((l) => l.includes("export const INPUT_ENDED_NOTICE")), head: headTs.find((l) => l.includes("export const INPUT_ENDED_NOTICE")) };
out.A7_exitCode_line = { base: baseTs.find((l) => /readonly exitCode:/.test(l))?.trim(), head: headTs.find((l) => /readonly exitCode:/.test(l))?.trim() };

out.A7_EVERY_TERMINAL_REASON = { base: block(baseTest, /^const EVERY_TERMINAL_REASON/, /\];/).text.split("\n").filter((l) => /^\s+"/.test(l)).length, head: block(headTest, /^const EVERY_TERMINAL_REASON/, /\];/).text.split("\n").filter((l) => /^\s+"/.test(l)).length };

// A9 — QE's own recount. Method: every `+` line of `git diff -U0` on src/turn-loop.ts, minus lines
// that are blank, or whose trimmed text begins with `*`, `/**`, `*/` or `//` (pure comment lines).
// Then, separately: how many of the survivors are RE-INDENTED copies of a `-` line (moved, not new).
const diff = git("diff", "-U0", `${BASE}..HEAD`, "--", "src/turn-loop.ts");
const plus = diff.split("\n").filter((l) => l.startsWith("+") && !l.startsWith("+++")).map((l) => l.slice(1));
const minus = diff.split("\n").filter((l) => l.startsWith("-") && !l.startsWith("---")).map((l) => l.slice(1));
const isComment = (l) => { const t = l.trim(); return t === "" || t.startsWith("*") || t.startsWith("/**") || t.startsWith("//") || t.startsWith("*/"); };
const addedCode = plus.filter((l) => !isComment(l));
const minusTrim = new Set(minus.map((l) => l.trim()));
const moved = addedCode.filter((l) => minusTrim.has(l.trim()));
const genuinelyNew = addedCode.filter((l) => !minusTrim.has(l.trim()));
out.A9 = { plusLines: plus.length, minusLines: minus.length, addedNonComment: addedCode.length, ofWhichReindentedMoves: moved.length, genuinelyNew: genuinelyNew.length, bound: 30, addedNonCommentList: addedCode, genuinelyNewList: genuinelyNew };
