// usage: node analyse.js <log.txt>   — paired per-round deltas vs base, round 0 discarded.
const fs = require("node:fs");
const rows = fs.readFileSync(process.argv[2], "utf8").split("\n").filter((l) => l.startsWith("round="));
const kv = (l) => Object.fromEntries(l.split(" |")[0].split(" ").map((p) => p.split("=")));
const by = {};
let bad = [];
for (const l of rows) {
  const r = kv(l);
  if (r.rc !== "0" || r.cli_deep_lines !== "0") bad.push(l);
  (by[r.round] ??= {})[r.tree] = { wall: +r.real, cpu: +r.user + +r.sys, load: +r.load1_before };
}
const med = (a) => { const s = [...a].sort((x, y) => x - y); const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };
const binom2 = (k, n) => { // exact two-sided sign test
  const c = (n, k) => { let r = 1; for (let i = 1; i <= k; i++) r = (r * (n - k + i)) / i; return r; };
  const lo = Math.min(k, n - k); let p = 0; for (let i = 0; i <= lo; i++) p += c(n, i); return Math.min(1, (2 * p) / 2 ** n);
};
const rounds = Object.keys(by).map(Number).filter((r) => r > 0).sort((a, b) => a - b);
const baseCpu = med(rounds.map((r) => by[r].base.cpu));
const baseWall = med(rounds.map((r) => by[r].base.wall));
console.log(`scored rounds n=${rounds.length}; base median CPU ${baseCpu.toFixed(2)} s, wall ${baseWall.toFixed(2)} s`);
console.log("round  load1(b/h/t)        dCPU head  dCPU twin  dWall head dWall twin");
for (const r of rounds) {
  const b = by[r].base, h = by[r].head, t = by[r].twin;
  console.log(`${String(r).padStart(5)}  ${[b, h, t].map((x) => x.load.toFixed(2)).join("/").padEnd(18)}  ${(h.cpu - b.cpu).toFixed(2).padStart(9)}  ${(t.cpu - b.cpu).toFixed(2).padStart(9)}  ${(h.wall - b.wall).toFixed(2).padStart(9)}  ${(t.wall - b.wall).toFixed(2).padStart(9)}`);
}
for (const [name, key] of [["twin", "twin"], ["head", "head"]]) {
  for (const inst of ["cpu", "wall"]) {
    const d = rounds.map((r) => by[r][key][inst] - by[r].base[inst]);
    const pos = d.filter((x) => x > 0).length, neg = d.filter((x) => x < 0).length;
    const m = med(d);
    const pct = (100 * m) / (inst === "cpu" ? baseCpu : baseWall);
    console.log(`${name}-base ${inst}: median ${m >= 0 ? "+" : ""}${m.toFixed(3)} s (${pct >= 0 ? "+" : ""}${pct.toFixed(2)} % of base), sign +${pos}/-${neg} of ${d.length}, two-sided p=${binom2(pos, pos + neg).toFixed(3)}`);
  }
}
const loads = rounds.flatMap((r) => ["base", "head", "twin"].map((t) => by[r][t].load));
console.log(`load1 range over scored runs: ${Math.min(...loads).toFixed(2)}–${Math.max(...loads).toFixed(2)}`);
console.log(bad.length ? `NON-GREEN OR CLI_DEEP RUNS:\n${bad.join("\n")}` : "every run (incl. round 0): rc=0, cli_deep_lines=0");
