// Count non-comment, non-blank production lines in the re-hop diff 5d0fdb0..85b66c6.
import { spawnSync } from "node:child_process";
const cwd = process.argv[2];
const out = {};
for (const f of ["src/command-input.ts", "src/line-input.ts"]) {
  const d = spawnSync("git", ["diff", "-U0", "5d0fdb0", "85b66c6", "--", f], { cwd, encoding: "utf8" }).stdout.split("\n");
  const code = (l) => !/^[+-]\s*(\*|\/\*\*|\/\/)/.test(l) && l.slice(1).trim().length > 0;
  const a = d.filter((l) => /^\+(?!\+\+ )/.test(l) && code(l));
  const r = d.filter((l) => /^-(?!-- )/.test(l) && code(l));
  out[f] = { add: a.length, del: r.length, added: a, removed: r };
}
console.log(JSON.stringify(out, null, 2));
