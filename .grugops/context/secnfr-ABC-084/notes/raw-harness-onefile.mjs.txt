// secnfr-ABC-084 supplement: CPU of ONE test file run alone (`npx vitest run <file>`), one fresh
// process per run, order rotated, round 0 discarded. P = src/puzzle-suite.test.ts (as built),
// Q = the remedy copy (A6 oracle row only under PUZZLE_DEEP), E = a one-assertion empty test file.
// P-E = the file's content net of the runner's per-process + per-file cost. Not the N8-LANE instrument.
import { execSync, spawnSync } from "node:child_process";
import { appendFileSync, writeFileSync } from "node:fs";
const DIR = process.env.DIR; const OUT = process.env.OUT; const ROUNDS = Number(process.env.ROUNDS ?? 13);
const F = { P: "src/puzzle-suite.test.ts", Q: "src/puzzle-suite.remedy.test.ts", E: "src/zz-empty.test.ts" };
const load1 = () => Number(execSync("sysctl -n vm.loadavg").toString().trim().split(/\s+/)[1]);
const C = Object.keys(F); const rows = [];
writeFileSync(OUT, "round\tcond\twall\tcpu\ttests\tload1\n");
for (let round = 0; round < ROUNDS; round++) {
  const k = round % C.length; const order = [...C.slice(k), ...C.slice(0, k)];
  for (const c of order) {
    const lb = load1();
    const r = spawnSync("/usr/bin/time", ["-p", "npx", "vitest", "run", F[c]], { cwd: DIR, encoding: "utf8" });
    const pick = (key) => Number((r.stderr.match(new RegExp(`^${key}\\s+([0-9.]+)`, "m")) ?? [])[1]);
    const t = Number(((r.stdout ?? "").match(/^\s*Tests\s+(\d+) passed/m) ?? [])[1]);
    const cpu = +(pick("user") + pick("sys")).toFixed(2);
    rows.push({ round, c, cpu, wall: pick("real"), t, lb, ok: r.status === 0 });
    appendFileSync(OUT, [round, c, pick("real"), cpu, t, lb].join("\t") + "\n");
  }
}
const med = (xs) => { const s = [...xs].sort((a, b) => a - b); const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };
const use = rows.filter((r) => r.round > 0);
for (const c of C) { const x = use.filter((r) => r.c === c); console.log(`${c} ${F[c]}: tests=${x[0].t} medianCPU=${med(x.map((r) => r.cpu))} range=${Math.min(...x.map((r) => r.cpu))}-${Math.max(...x.map((r) => r.cpu))} medianWALL=${med(x.map((r) => r.wall))}`); }
for (const [x, y] of [["E", "P"], ["E", "Q"], ["Q", "P"]]) {
  const d = []; for (let round = 1; round < ROUNDS; round++) d.push(+(use.find((r) => r.round === round && r.c === y).cpu - use.find((r) => r.round === round && r.c === x).cpu).toFixed(2));
  console.log(`${y}-${x} CPU per round: ${d.join(" ")} median=${med(d).toFixed(3)} sign+ ${d.filter((v) => v > 0).length}/${d.length}`);
}
console.log(`load1: min=${Math.min(...rows.map((r) => r.lb))} median=${med(rows.map((r) => r.lb))} max=${Math.max(...rows.map((r) => r.lb))}; failures=${rows.filter((r) => !r.ok).length}`);
