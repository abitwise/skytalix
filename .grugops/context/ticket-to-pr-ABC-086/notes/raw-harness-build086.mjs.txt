import { execFileSync } from "node:child_process";
import { mkdirSync, rmSync, symlinkSync } from "node:fs";

const S = "/private/tmp/claude-501/-Users-olgeroeselg-Projects-hacks-grugops-examples-cli-chess-example/e0dd4cf5-83f1-4fb6-bb33-18c0c2394948/scratchpad/n8/arms";
const W = "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/ABC-086-virtual-clock";
const NM = "/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/node_modules";
const git = (...a) => execFileSync("git", ["-C", W, ...a], { maxBuffer: 256 * 1024 * 1024 });

rmSync(S, { recursive: true, force: true });
for (const [arm, rev] of [["A", "1ae5284"], ["N", "1ae5284"], ["B", "1d74693"]]) {
  mkdirSync(`${S}/${arm}`, { recursive: true });
  execFileSync("tar", ["-x", "-C", `${S}/${arm}`], { input: git("archive", rev) });
  symlinkSync(NM, `${S}/${arm}/node_modules`);
}
