function vitest(args, tag) {
  const outFile = `${OUT}/raw-mutant-${tag}.json`;
  const t0 = Date.now();
  const r = spawnSync("npx", ["vitest", "run", ...args, "--reporter=json", `--outputFile=${outFile}`], {
    cwd: CWD, encoding: "utf8", maxBuffer: 1 << 28,
  });
  const ms = Date.now() - t0;
  let j;
  try { j = JSON.parse(readFileSync(outFile, "utf8")); } catch (e) { return { exit: r.status, ms, parseError: String(e), stderr: (r.stderr ?? "").slice(-2000) }; }
  const red = [];
  for (const f of j.testResults) {
    for (const a of f.assertionResults) {
      if (a.status === "failed") red.push({ file: f.name.replace(`${CWD}/`, ""), name: a.fullName, msg: (a.failureMessages?.[0] ?? "").split("\n")[0].slice(0, 220) });
    }
  }
  return { exit: r.status, ms, total: j.numTotalTests, passed: j.numPassedTests, failed: j.numFailedTests, pending: j.numPendingTests, files: j.testResults.length, red };
}

const original = readFileSync(FILE, "utf8");
const results = [];
for (const m of MUTANTS) {
  if (only.length && !only.includes(m.id)) continue;
  let src = original;
  for (const [from, to] of m.edits) {
    const n = src.split(from).length - 1;
    if (n !== 1) throw new Error(`mutant ${m.id}: patch anchor matched ${n} times: ${from.slice(0, 60)}`);
    src = src.replace(from, to);
  }
  if (src === original) throw new Error(`mutant ${m.id}: no change`);
  writeFileSync(FILE, src);
  const diff = git("diff", "--numstat", "--", "src/zobrist.ts").trim();
  const l = load();
  const two = vitest(["src/zobrist.test.ts", "src/zobrist.nfr.test.ts"], `${m.id}-two`);
  let lane = null;
  if (two.red && two.red.length === 0) lane = vitest(["--dir", "src"], `${m.id}-lane`);
  git("checkout", "--", "src/zobrist.ts");
  const clean = git("status", "--porcelain", "--", "src").trim();
  if (clean !== "") throw new Error(`src not clean after mutant ${m.id}: ${clean}`);
  if (readFileSync(FILE, "utf8") !== original) throw new Error(`mutant ${m.id}: revert did not restore bytes`);
  const rec = { id: m.id, what: m.what, numstat: diff, load: l, twoFiles: two, wholeLane: lane, revertedClean: true };
  results.push(rec);
  console.log(`[${m.id}] ${diff} | two-files: ${two.failed} failed / ${two.passed} passed in ${two.ms} ms${lane ? ` | LANE: ${lane.failed} failed / ${lane.passed} passed` : ""}`);
  for (const r of two.red ?? []) console.log(`    RED ${r.name} :: ${r.msg}`);
}
writeFileSync(`${OUT}/mutants-summary${only.length ? "-" + only.join("") : ""}.json`, JSON.stringify(results, null, 2));
