const show = (label: string, fen: string, line: string) => {
  const h = newGame(fen);
  const before = toFen(h.position);
  const out = resolveMoveInput(h, line);
  const msgs = out.kind === "move" ? `[move ${squareName(out.move.from)}${squareName(out.move.to)}]` : JSON.stringify(out.messages);
  const rej = out.kind === "rejected" ? classifyRejection(h.position, [line]).reason : "-";
  console.log(`${label.padEnd(34)} ${JSON.stringify(line).padEnd(14)} -> ${out.kind.padEnd(9)} ${msgs}  classifier(raw)=${rej}  fenSame=${toFen(h.position) === before}`);
};

console.log("== A5 in ASCII mode and both orientations, all nine sentences, hidden count ==");
const NINE: [string, string][] = [[START, "hello"], [START, "e3e4"], [START, "e7e5"], [START, "e2e5"], [PIN, "Nc3"], [CHECK, "Nh3"], [START, "Qh5"], [PROMO, "e8=K"], [NOCASTLE, "O-O"], [START, "Qa1xh8#"], [START, "  Qh5  "]];

    for (const [fen, line] of NINE) {
      const h = newGame(fen);
      const out = resolveMoveInput(h, line) as any;
      const frame: string[] = assembleFrame(
        { board: renderBoard(h.position, glyphs, orient), status: renderStatusPanel(h, gameStatusOf(h), false), history: renderHistory(h), messages: out.messages },
        glyphs,
      );
      const region = frame.slice(frame.length - PROMPT_ROWS - MESSAGE_ROWS, frame.length - PROMPT_ROWS);
      const hidden = /\(\+(\d+) more\)/.exec(region.join("\n"));
      const whole = region.map((r) => r.trimEnd()).includes(out.messages[0]);
      console.log(`  ${glyphs.padEnd(7)} ${orient.padEnd(12)} ${JSON.stringify(line).padEnd(10)} "${out.messages[0]}" (${out.messages[0].length} cols) hidden=${hidden ? hidden[1] : 0} whole=${whole} rows=${frame.length}`);
    }
  }
