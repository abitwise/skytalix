#!/bin/bash
# oot.sh <name> <node-script-or-"-"> <PUZZLE_DEEP value> [vitest -t filter]
# Copies the worktree's working tree (tracked + untracked, not ignored) out of tree, applies an
# optional plant script (node <script> <copy-dir>), runs the deep lane there, logs load1 + exit.
set -u
S=/private/tmp/claude-501/-Users-olgeroeselg-Projects-hacks-grugops-examples-cli-chess-example/e0dd4cf5-83f1-4fb6-bb33-18c0c2394948/scratchpad/abc131
WT=/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/.claude/worktrees/ticket-to-pr-ABC-131
name=$1; plant=$2; mode=$3; filt=${4:-}
D=$S/copies/$name; rm -rf "$D"; mkdir -p "$D"
(cd "$WT" && git ls-files -co --exclude-standard -- src package.json package-lock.json tsconfig*.json vitest.config.* biome.json* | tar -cf - -T -) | tar -xf - -C "$D"
ln -s /Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example/node_modules "$D/node_modules"
if [ "$plant" != "-" ]; then node "$plant" "$D" || { echo "PLANT FAILED $name"; exit 2; }; fi
O=$S/out/$name.txt
{ echo "RUN $name UTC $(date -u +%FT%TZ) load $(sysctl -n vm.loadavg) wt-head $(git -C "$WT" rev-parse --short HEAD) PUZZLE_DEEP=$mode filter=${filt:-none}"
  if [ -n "$filt" ]; then (cd "$D" && /usr/bin/time -p env PUZZLE_DEEP=$mode npx vitest run src/puzzle-suite.test.ts -t "$filt" 2>&1); else (cd "$D" && /usr/bin/time -p env PUZZLE_DEEP=$mode npx vitest run src/puzzle-suite.test.ts 2>&1); fi
  echo "EXIT $?"; echo "END UTC $(date -u +%FT%TZ) load $(sysctl -n vm.loadavg)"; } > "$O" 2>&1
grep -E "^(SOLVED|MISSED)" "$O" > "$S/out/$name.rows.txt"
echo "== $name: $(grep -E '^EXIT' $O) | $(grep -E '^ +Tests ' $O | head -1 | xargs) | $(grep -E '^PUZZLES' $O) | $(grep -E '^real' $O)"
grep -E "^NEWLY|AssertionError|^ +FAIL |✗|×" "$O" | head -8
