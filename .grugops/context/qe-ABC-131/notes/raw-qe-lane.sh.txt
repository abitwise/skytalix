#!/bin/bash
# lane.sh <name> <commit> <ENV=VAL> <test file>  -- git-archive copy, out of tree, load1 logged.
set -u
Q=/private/tmp/claude-501/-Users-olgeroeselg-Projects-hacks-grugops-examples-cli-chess-example/e0dd4cf5-83f1-4fb6-bb33-18c0c2394948/scratchpad/qe131
REPO=/Users/olgeroeselg/Projects/hacks/grugops-examples/cli-chess-example
name=$1; commit=$2; envkv=$3; file=$4
D=$Q/copies/$name; rm -rf "$D"; mkdir -p "$D"
git -C "$REPO" archive "$commit" src package.json tsconfig.json vitest.config.ts | tar -x -C "$D"
ln -s "$REPO/node_modules" "$D/node_modules"
O=$Q/out/$name.txt
{ echo "RUN $name commit $(git -C "$REPO" rev-parse --short "$commit") $envkv $file UTC $(date -u +%FT%TZ) load $(sysctl -n vm.loadavg)"
  (cd "$D" && /usr/bin/time -p env $envkv npx vitest run "$file" 2>&1); echo "EXIT $?"
  echo "END UTC $(date -u +%FT%TZ) load $(sysctl -n vm.loadavg)"; } > "$O" 2>&1
echo "== $name: $(grep -E '^EXIT' $O) | $(grep -E '^ +Tests ' $O | head -1 | xargs) | $(grep -E '^real' $O) | start $(head -1 $O | sed 's/.*load //')"
rm -rf "$D"
